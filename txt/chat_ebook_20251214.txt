### 3.2 查询所有使用文号的表单（SQL查询）
#### 方法1：通过govdoc_mark_record查询
```sql
-- 查询所有使用过文号的表单及统计
SELECT
t.id AS template_id,
t.subject AS template_name,
t.app_name,
COUNT(DISTINCT gmr.id) AS mark_count,
COUNT(DISTINCT gmr.summary_id) AS summary_count,
COUNT(DISTINCT CASE WHEN gmr.newflow_type = 0 THEN gmr.summary_id END) AS main_flow_count,
COUNT(DISTINCT CASE WHEN gmr.newflow_type = 1 THEN gmr.summary_id END) AS child_flow_count,
COUNT(DISTINCT CASE WHEN gmr.newflow_type = 2 THEN gmr.summary_id END) AS auto_flow_count
FROM ctp_template t
INNER JOIN edoc_summary es ON es.templete_id = t.id
INNER JOIN govdoc_mark_record gmr ON gmr.summary_id = es.id
WHERE gmr.used_state = 1 -- 1表示已使用
GROUP BY t.id, t.subject, t.app_name
ORDER BY mark_count DESC;
```
#### 方法2：通过edoc_mark_history查询（已完成的文号）
```sql
-- 查询已完成流程中使用文号的表单统计
SELECT
t.id AS template_id,
t.subject AS template_name,
t.app_name,
COUNT(DISTINCT emh.id) AS history_mark_count,
COUNT(DISTINCT emh.edoc_id) AS completed_summary_count,
MIN(emh.create_time) AS first_use_time,
MAX(emh.complete_time) AS last_use_time
FROM ctp_template t
INNER JOIN edoc_summary es ON es.templete_id = t.id
INNER JOIN edoc_mark_history emh ON emh.edoc_id = es.id
GROUP BY t.id, t.subject, t.app_name
ORDER BY history_mark_count DESC;
```
#### 方法3：查询特定时间段内使用文号的表单
```sql
-- 查询最近一个月使用文号的表单
SELECT
t.id AS template_id,
t.subject AS template_name,
md.word_no AS mark_word,
mc.name AS category_name,
COUNT(DISTINCT gmr.summary_id) AS use_count,
MAX(gmr.create_time) AS last_use_time
FROM ctp_template t
INNER JOIN edoc_summary es ON es.templete_id = t.id
INNER JOIN govdoc_mark_record gmr ON gmr.summary_id = es.id
INNER JOIN edoc_mark_definition md ON md.id = gmr.mark_def_id
INNER JOIN edoc_mark_category mc ON mc.id = md.category_id
WHERE gmr.create_time >= DATE_SUB(NOW(), INTERVAL 1 MONTH)
GROUP BY t.id, t.subject, md.word_no, mc.name
ORDER BY use_count DESC;
```

## 四、如何统计所有表单使用文号情况
### 4.1 综合统计SQL
```sql
-- 全面统计所有表单的文号使用情况
SELECT
t.id AS template_id,
t.subject AS template_name,
t.app_name,
-- 文号使用统计
COUNT(DISTINCT gmr.id) AS total_mark_records,
COUNT(DISTINCT CASE WHEN gmr.used_state = 1 THEN gmr.id END) AS used_marks,
COUNT(DISTINCT CASE WHEN gmr.used_state = 0 THEN gmr.id END) AS unused_marks,
-- 按流程类型统计
COUNT(DISTINCT CASE WHEN gmr.newflow_type = 0 THEN gmr.summary_id END) AS main_flow_count,
COUNT(DISTINCT CASE WHEN gmr.newflow_type = 1 THEN gmr.summary_id END) AS child_flow_count,
COUNT(DISTINCT CASE WHEN gmr.newflow_type = 2 THEN gmr.summary_id END) AS auto_flow_count,
-- 按公文类型统计
COUNT(DISTINCT CASE WHEN gmr.govdoc_type = 1 THEN gmr.id END) AS send_doc_count,
COUNT(DISTINCT CASE WHEN gmr.govdoc_type = 2 THEN gmr.id END) AS receive_doc_count,
COUNT(DISTINCT CASE WHEN gmr.govdoc_type = 3 THEN gmr.id END) AS sign_doc_count,
-- 时间统计
MIN(gmr.create_time) AS first_use_time,
MAX(gmr.last_time) AS last_use_time,
-- 文号定义信息
COUNT(DISTINCT gmr.mark_def_id) AS mark_def_count,
GROUP_CONCAT(DISTINCT md.word_no ORDER BY md.word_no SEPARATOR ', ') AS mark_words
FROM ctp_template t
INNER JOIN edoc_summary es ON es.templete_id = t.id
INNER JOIN govdoc_mark_record gmr ON gmr.summary_id = es.id
LEFT JOIN edoc_mark_definition md ON md.id = gmr.mark_def_id
WHERE gmr.create_time >= '2024-01-01' -- 可调整时间范围
GROUP BY t.id, t.subject, t.app_name
ORDER BY total_mark_records DESC;
```
### 4.2 按文号类别统计
```sql
-- 按文号类别统计使用情况
SELECT
mc.id AS category_id,
mc.name AS category_name,
mc.code_mode,
COUNT(DISTINCT gmr.id) AS mark_count,
COUNT(DISTINCT gmr.summary_id) AS summary_count,
COUNT(DISTINCT gmr.mark_def_id) AS def_count,
COUNT(DISTINCT t.id) AS template_count,
MIN(gmr.create_time) AS first_use_time,
MAX(gmr.last_time) AS last_use_time
FROM edoc_mark_category mc
INNER JOIN edoc_mark_definition md ON md.category_id = mc.id
INNER JOIN govdoc_mark_record gmr ON gmr.mark_def_id = md.id
INNER JOIN edoc_summary es ON es.id = gmr.summary_id
INNER JOIN ctp_template t ON t.id = es.templete_id
GROUP BY mc.id, mc.name, mc.code_mode
ORDER BY mark_count DESC;
```
### 4.3 按年度统计文号使用情况
```sql
-- 按年度统计文号使用情况
SELECT
YEAR(gmr.create_time) AS use_year,
gmr.year_no,
COUNT(DISTINCT gmr.id) AS mark_count,
COUNT(DISTINCT gmr.summary_id) AS summary_count,
COUNT(DISTINCT t.id) AS template_count,
COUNT(DISTINCT CASE WHEN gmr.newflow_type = 0 THEN gmr.id END) AS main_flow_count,
COUNT(DISTINCT CASE WHEN gmr.newflow_type = 1 THEN gmr.id END) AS child_flow_count
FROM govdoc_mark_record gmr
INNER JOIN edoc_summary es ON es.id = gmr.summary_id
INNER JOIN ctp_template t ON t.id = es.templete_id
WHERE gmr.used_state = 1
GROUP BY YEAR(gmr.create_time), gmr.year_no
ORDER BY use_year DESC, gmr.year_no;
```
### 4.4 查询文号重复使用情况
```sql
-- 检测文号重复使用（可能存在的问题）
SELECT
gmr.markstr AS mark_number,
COUNT(DISTINCT gmr.summary_id) AS use_count,
GROUP_CONCAT(DISTINCT es.subject ORDER BY es.create_time SEPARATOR ' | ') AS subjects,
MIN(gmr.create_time) AS first_use,
MAX(gmr.last_time) AS last_use
FROM govdoc_mark_record gmr
INNER JOIN edoc_summary es ON es.id = gmr.summary_id
WHERE gmr.markstr IS NOT NULL
AND gmr.markstr != ''
GROUP BY gmr.markstr
HAVING COUNT(DISTINCT gmr.summary_id) > 1
ORDER BY use_count DESC;
```

```sql
GROUP BY gmr.markstr
```

### 2.1 Hibernate type=\"string\" vs type=\"text\"

## 八、根因总结
### 问题链条
```
1. Hibernate 映射使用 type=\"string\"
↓
2. JDBC 使用 ResultSet.getString() 读取 CLOB
↓
3. 某些驱动对 getString() 有隐式长度限制
↓
4. 超过限制后返回的字符串被截断
↓
5. Jackson 解析被截断的 JSON
↓
6. 抛出 \"Unexpected end-of-input\" 异常
```

1. **立即**: 修改 Hibernate 映射为 `type=\"text\"`（方案1）

### 问题原理
```
Hibernate type="string"
↓

↓
```

**而 `type="text"`**:
- 使用 `ResultSet.getClob()` 或 `getCharacterStream()`
- 流式读取，**无长度限制**
- 正确处理大文本

```java
* CaseContent 写入验证工具
* 用于排查 JSON 序列化和写入问题
*/
public class CaseContentValidator {
private static final Log logger = CtpLogFactory.getLog(CaseContentValidator.class);
// JSON 长度阈值（根据实际情况调整）
private static final int WARN_THRESHOLD = 100000; // 10万字符警告
private static final int ERROR_THRESHOLD = 140000; // 14万字符告警
private static final int MAX_SAFE_LENGTH = 200000; // 20万字符最大安全值
/**
* 验证 JSON 字符串完整性
*/
public static boolean isValidJson(String json) {
if (json == null || json.isEmpty()) {
return false;
}
try {
// 检查括号匹配
int openBraces = 0;
int closeBraces = 0;
int openBrackets = 0;
int closeBrackets = 0;
for (char c : json.toCharArray()) {
if (c == '{') openBraces++;
if (c == '}') closeBraces++;
if (c == '[') openBrackets++;
if (c == ']') closeBrackets++;
}
boolean balanced = (openBraces == closeBraces) && (openBrackets == closeBrackets);
if (!balanced) {
logger.error(String.format(\"JSON 括号不匹配: { %d/%d, [ %d/%d\",
openBraces, closeBraces, openBrackets, closeBrackets));
}
return balanced;
} catch (Exception e) {
logger.error(\"JSON 验证失败\", e);
return false;
}
}

*/
public static ValidationResult validateBeforeSave(BPMCaseObject caseObject,
long caseId,
String processId) {

result.processId = processId;
if (caseObject == null) {
result.valid = false;
result.errorMessage = \"BPMCaseObject 为 null\";
logger.error(result.toString());
return result;
}
// 统计数据量
List<BPMCaseLog> caseLogs = caseObject.getCaseLog();
if (caseLogs != null) {
result.caseLogCount = caseLogs.size();
// 统计详细日志
int totalDetailLogs = 0;
for (BPMCaseLog log : caseLogs) {
if (log.getDetailLogList() != null) {
totalDetailLogs += log.getDetailLogList().size();
}
}
result.detailLogCount = totalDetailLogs;
}
// 序列化生成 JSON
String json = null;
long startTime = System.currentTimeMillis();
try {
json = JSONUtil.toJSONString(caseObject);
result.jsonGenerateTime = System.currentTimeMillis() - startTime;
} catch (Exception e) {
result.valid = false;
result.errorMessage = \"JSON 序列化失败: \" + e.getMessage();
logger.error(result.toString(), e);
return result;
}
// 检查 JSON 长度
if (json != null) {
result.jsonLength = json.length();
result.jsonCharLength = json.toCharArray().length;
if (result.jsonLength > MAX_SAFE_LENGTH) {
result.valid = false;
result.errorMessage = String.format(\"JSON 长度超过最大安全值 (%d > %d)\",
result.jsonLength, MAX_SAFE_LENGTH);
logger.error(result.toString());
return result;
} else if (result.jsonLength > ERROR_THRESHOLD) {
result.riskLevel = \"HIGH\";
logger.error(String.format(\"HIGH RISK: JSON 接近截断阈值: %s\", result));
} else if (result.jsonLength > WARN_THRESHOLD) {
result.riskLevel = \"MEDIUM\";
logger.warn(String.format(\"MEDIUM RISK: JSON 超过警告阈值: %s\", result));
}
// 验证 JSON 完整性
result.jsonComplete = isValidJson(json);
if (!result.jsonComplete) {
result.valid = false;
result.errorMessage = \"JSON 生成后不完整（括号不匹配）\";
logger.error(result.toString());
}
}
// 记录详细信息
if (result.jsonLength > WARN_THRESHOLD || !result.valid) {
logger.info(result.toDetailString());
}
return result;
}

result.processId = processId;

logger.error(result.toString());

}
result.jsonLength = savedJson.length();
```

```java
* 安全地设置 CaseContent（带验证和降级）
*/
private void safeSaveWithManyForms CaseContent(BPMCase bpmCase) throws BPMException {
if (bpmCase == null || bpmCase.getCaseObject() == null) {
logger.warn(\"BPMCase 或 CaseObject 为 null，跳过保存\");
return;
}
BPMCaseObject caseObject = bpmCase.getCaseObject();
long caseId = bpmCase.getId();
String processId = bpmCase.getProcessId();
// 1. 保存前验证
ValidationResult validation = CaseContentValidator.validateBeforeSave(
caseObject, caseId, processId);
if (!validation.valid) {
logger.error(\"CaseContent 验证失败: \" + validation);
// 继续尝试保存，但记录错误
}
String jsonContent = null;
try {
// 2. 序列化为 JSON
jsonContent = JSONUtil.toJSONString(caseObject);
// 3. 二次验证（防止序列化过程中出错）
if (!CaseContentValidator.isValidJson(jsonContent)) {
logger.error(String.format(\"JSON 序列化后不完整: caseId=%d, length=%d\",
caseId, jsonContent.length()));
}
// 4. 同时保存两种格式
// 4.1 优先使用二进制格式（BLOB）
try {
byte[] caseObjectBytes = WorkflowUtil.getBlobFromObject(caseObject.toMap());
if (bpmCase instanceof CaseRunDAO) {
((CaseRunDAO) bpmCase).setCaseObject2(caseObjectBytes);
} else if (bpmCase instanceof HistoryCaseRunDAO) {
((HistoryCaseRunDAO) bpmCase).setCaseObject2(caseObjectBytes);
}
logger.debug(String.format(\"CaseObject2 (BLOB) 保存成功: caseId=%d, size=%d bytes\",
caseId, caseObjectBytes.length));
} catch (Exception e) {
logger.error(\"保存 caseObject2 (BLOB) 失败，仅使用 JSON 格式\", e);
}
// 4.2 保存 JSON 格式（文本）
if (bpmCase instanceof CaseRunDAO) {
((CaseRunDAO) bpmCase).setCaseContent(jsonContent);
} else if (bpmCase instanceof HistoryCaseRunDAO) {
((HistoryCaseRunDAO) bpmCase).setCaseContent(jsonContent);
}
logger.info(String.format(\"CaseContent 保存: caseId=%d, processId=%s, jsonLength=%d, risk=%s\",
caseId, processId, jsonContent.length(), validation.riskLevel));
} catch (Exception e) {
logger.error(String.format(\"序列化 CaseContent 失败: caseId=%d, processId=%s\",
caseId, processId), e);
throw new BPMException(\"保存流程实例数据失败\", e);
}
}
```
